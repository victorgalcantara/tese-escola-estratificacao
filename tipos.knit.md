(function () {
  var estado = { f: "enem", v: "tabela" };
  var paineis = document.querySelectorAll(".painel");
  var botoesF = document.querySelectorAll("#seg-filtro button");
  var botoesV = document.querySelectorAll("#seg-vista button");
  function atualizar() {
    paineis.forEach(function (p) {
      p.classList.toggle("ativo", p.getAttribute("data-f") === estado.f && p.getAttribute("data-v") === estado.v);
    });
    botoesF.forEach(function (b) { b.setAttribute("aria-pressed", String(b.getAttribute("data-f") === estado.f)); });
    botoesV.forEach(function (b) { b.setAttribute("aria-pressed", String(b.getAttribute("data-v") === estado.v)); });
    // graficos criados com o painel oculto precisam recalcular a largura
    setTimeout(function () { window.dispatchEvent(new Event("resize")); }, 60);
    try { history.replaceState(null, "", "#" + estado.f + "-" + estado.v); } catch (e) { /* ignora */ }
  }
  botoesF.forEach(function (b) { b.addEventListener("click", function () { estado.f = b.getAttribute("data-f"); atualizar(); }); });
  botoesV.forEach(function (b) { b.addEventListener("click", function () { estado.v = b.getAttribute("data-v"); atualizar(); }); });
  var m = location.hash.match(/^#(enem|acesso|retornos)-(tabela|grafico)$/);
  if (m) { estado.f = m[1]; estado.v = m[2]; }
  atualizar();
})();
